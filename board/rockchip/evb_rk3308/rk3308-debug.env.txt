# Default environment for RK3308 debug builds (rk3308-debug.config).
# No bootcmd: U-Boot stops at the prompt. Only the load addresses (same as
# ENV_MEM_LAYOUT_SETTINGS in include/configs/rk3308_common.h, all below the
# OP-TEE TZDRAM at 0x0a000000) and what extlinux / bootstd need, so a boot
# can be started by hand:
#   => bootflow scan -lb mmc1          (SD card)
# This file is not preprocessed: fdtfile is the ROCK Pi S one, on the
# ROCK S0 use "setenv fdtfile rockchip/rk3308-rock-s0.dtb".

script_size_f=0x2000
pxefile_addr_r=0x00600000
fdt_addr_r=0x01e00000
fdtoverlay_addr_r=0x01f00000
kernel_addr_r=0x02080000
ramdisk_addr_r=0x06000000
kernel_comp_addr_r=0x08000000
kernel_comp_size=0x2000000
fdtfile=rockchip/rk3308-rock-pi-s.dtb
boot_targets=mmc1 mmc0
